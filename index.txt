    <script>
        class AIExcelAnalyzer {
            constructor() {
                this.data = [];
                this.names = [];
                this.init();
            }

            init() {
                const uploadBox = document.getElementById('uploadBox');
                const fileInput = document.getElementById('fileInput');
                const analyzeBtn = document.getElementById('analyzeBtn');

                // Drag & Drop
                uploadBox.addEventListener('dragover', (e) => {
                    e.preventDefault();
                    uploadBox.classList.add('dragover');
                });

                uploadBox.addEventListener('dragleave', () => {
                    uploadBox.classList.remove('dragover');
                });

                uploadBox.addEventListener('drop', (e) => {
                    e.preventDefault();
                    uploadBox.classList.remove('dragover');
                    const files = e.dataTransfer.files;
                    if (files.length > 0) {
                        fileInput.files = files;
                        this.handleFile(files[0]);
                    }
                });

                uploadBox.addEventListener('click', () => fileInput.click());

                fileInput.addEventListener('change', (e) => {
                    if (e.target.files.length > 0) {
                        this.handleFile(e.target.files[0]);
                    }
                });

                analyzeBtn.addEventListener('click', () => this.analyze());
            }

            handleFile(file) {
                const uploadBox = document.getElementById('uploadBox');
                const analyzeBtn = document.getElementById('analyzeBtn');
                
                uploadBox.innerHTML = `
                    <div class="upload-icon">✅</div>
                    <h3>${file.name}</h3>
                    <p>تم تحميل الملف بنجاح!</p>
                    <button class="btn" id="analyzeBtn" style="background: #4ecdc4;">ابدأ التحليل الذكي</button>
                `;
                
                analyzeBtn = document.getElementById('analyzeBtn');
                analyzeBtn.disabled = false;

                const reader = new FileReader();
                reader.onload = (e) => {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                    this.data = XLSX.utils.sheet_to_json(firstSheet);
                    analyzeBtn.onclick = () => this.analyze();
                };
                reader.readAsArrayBuffer(file);
            }

            extractNames(text) {
                // قائمة أسماء عربية شائعة
                const arabicNames = [
                    'محمد', 'أحمد', 'علي', 'عمر', 'خالد', 'يوسف', 'عبدالله', 'إبراهيم', 
                    'حسن', 'حسين', 'سعيد', 'مصطفى', 'فاطمة', 'مريم', 'عائشة', 'زينب',
                    'نور', 'لين', 'ريم', 'سارة', 'هدى', 'منى', 'ليلى', 'نسمة'
                ];

                const words = text.toString().split(/\s+/);
                const foundNames = [];

                words.forEach(word => {
                    const cleanWord = word.replace(/[^\u0600-\u06FF\u0750-\u077Fa-zA-Z]/g, '').trim();
                    if (cleanWord.length > 2 && arabicNames.some(name => 
                        cleanWord.includes(name) || name.includes(cleanWord))) {
                        foundNames.push(cleanWord);
                    }
                });

                return [...new Set(foundNames)];
            }

            analyze() {
                document.getElementById('upload-area').style.display = 'none';
                document.getElementById('loading').style.display = 'block';

                setTimeout(() => {
                    this.processData();
                    this.showResults();
                    document.getElementById('loading').style.display = 'none';
                    document.getElementById('results').style.display = 'block';
                }, 2000);
            }

            processData() {
                this.names = [];
                
                this.data.forEach((row, index) => {
                    Object.values(row).forEach(cell => {
                        if (cell && typeof cell === 'string') {
                            const foundNames = this.extractNames(cell);
                            foundNames.forEach(name => {
                                this.names.push({
                                    name: name,
                                    row: index + 1,
                                    frequency: 1
                                });
                            });
                        }
                    });
                });

                // حساب التكرار
                const nameFreq = {};
                this.names.forEach(item => {
                    nameFreq[item.name] = (nameFreq[item.name] || 0) + 1;
                });

                this.nameStats = Object.entries(nameFreq)
                    .map(([name, freq]) => ({ name, frequency: freq }))
                    .sort((a, b) => b.frequency - a.frequency);
            }

            showResults() {
                this.showNameStats();
                this.showGeneralStats();
                this.showNamesList();
                this.showCharts();
                this.generateAIInsights();
            }

            showNameStats() {
                const statsHtml = this.nameStats.slice(0, 6).map(stat => `
                    <div class="stat-item">
                        <div class="stat-number">${stat.frequency}</div>
                        <div class="stat-label">${stat.name}</div>
                    </div>
                `).join('');

                document.getElementById('nameStats').innerHTML = statsHtml;
            }

            showGeneralStats() {
                const totalRows = this.data.length;
                const totalNames = this.names.length;
                const uniqueNames = this.nameStats.length;
                const avgFreq = totalNames > 0 ? (totalNames / uniqueNames).toFixed(1) : 0;

                document.getElementById('generalStats').innerHTML = `
                    <div class="stat-item">
                        <div class="stat-number">${totalRows}</div>
                        <div class="stat-label">عدد الصفوف</div>
                    </div>
                    <div class="stat-item">
                        <div class="stat-number">${totalNames}</div>
                        <div class="stat-label">إجمالي الأسماء</div>
                    </div>
                    <div class="stat-item">
                        <div class="stat-number">${uniqueNames}</div>
                        <div class="stat-label">الأسماء الفريدة</div>
                    </div>
                    <div class="stat-item">
                        <div class="stat-number">${avgFreq}</div>
                        <div class="stat-label">متوسط التكرار</div>
                    </div>
                `;
            }

            showNamesList() {
                const listHtml = this.nameStats.slice(0, 20).map(stat => `
                    <div class="name-item">
                        <span>${stat.name}</span>
                        <span style="color: #667eea; font-weight: bold;">${stat.frequency} مرة</span>
                    </div>
                `).join('');

                document.getElementById('namesList').innerHTML = listHtml || '<div style="text-align:center;color:#999;">لا توجد أسماء</div>';
            }

            showCharts() {
                const ctx = document.getElementById('nameChart').getContext('2d');
                new Chart(ctx, {
                    type: 'bar',
                    data: {
                        labels: this.nameStats.slice(0, 10).map(s => s.name),
                        datasets: [{
                            label: 'تكرار الاسم',
                            data: this.nameStats.slice(0, 10).map(s => s.frequency),
                            backgroundColor: 'rgba(102, 126, 234, 0.7)',
                            borderColor: 'rgba(102, 126, 234, 1)',
                            borderWidth: 2
                        }]
                    },
                    options: {
                        responsive: true,
                        maintainAspectRatio: false,
                        scales: {
                            y: {
                                beginAtZero: true
                            }
                        },
                        plugins: {
                            legend: {
                                display: false
                            }
                        }
                    }
                });
            }

            generateAIInsights() {
                const insights = [];
                
                if (this.nameStats.length > 0) {
                    const topName = this.nameStats[0];
                    insights.push(`الاسم الأكثر تكراراً: "${topName.name}" (${topName.frequency} مرة)`);
                    
                    const top3 = this.nameStats.slice(0, 3).map(s => s.name).join('، ');
                    insights.push(`أكثر 3 أسماء شيوعاً: ${top3}`);
                }

                if (this.data.length > 50) {
                    insights.push('البيانات كبيرة الحجم - يُنصح بتقسيمها لتحليل أدق');
                }

                document.getElementById('aiInsights').innerHTML = insights.join('<br>');
            }
        }

        // تشغيل التطبيق
        new AIExcelAnalyzer();
    </script>
